MainController.java 
// Define que essa classe pertence ao pacote "controller"
package com.template.controller;

// Importa o DAO, que é responsável por conversar com o banco de dados
import com.template.model.dao.LivroDAO;

// Importa o DTO, que representa os dados de um livro
import com.template.model.dto.LivroDTO;

// Importa o Validator, responsável por verificar se os dados são válidos
import com.template.validator.LivroValidator;

// Classes do JavaFX usadas para trabalhar com listas
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

// Usado para representar eventos, como clicar em um botão
import javafx.event.ActionEvent;

// Permite conectar os elementos do FXML com o Java
import javafx.fxml.FXML;

// Componentes visuais do JavaFX
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;

// Permite definir qual atributo do objeto aparece em cada coluna da tabela
import javafx.scene.control.cell.PropertyValueFactory;


// Controller da tela principal
public class MainController {

    // @FXML conecta essas variáveis aos elementos que estão no arquivo FXML

    // Campo onde o usuário digita o ID do livro
    @FXML private TextField txtId;

    // Campo onde o usuário digita o título
    @FXML private TextField txtTitulo;

    // Campo onde o usuário digita o autor
    @FXML private TextField txtAutor;

    // Campo onde o usuário digita o ano de publicação
    @FXML private TextField txtAno;

    // Label usada para mostrar mensagens na tela
    @FXML private Label lblMensagem;

    // Tabela que mostra os livros
    @FXML private TableView<LivroDTO> tabelaLivros;

    // Colunas da tabela
    @FXML private TableColumn<LivroDTO, Integer> colId;
    @FXML private TableColumn<LivroDTO, String> colTitulo;
    @FXML private TableColumn<LivroDTO, String> colAutor;
    @FXML private TableColumn<LivroDTO, Integer> colAno;

    // Botões da tela
    @FXML private Button btnCadastrar;
    @FXML private Button btnListar;
    @FXML private Button btnApagar;
    @FXML private Button btnAtualizar;
    @FXML private Button btnLimpar;
    @FXML private Button btnBuscar;


    // Cria um objeto LivroDAO.
    // Ele será usado para cadastrar, buscar, atualizar e excluir livros no banco.
    private final LivroDAO livroDAO = new LivroDAO();


    // Método executado automaticamente quando a tela é inicializada
    @FXML
    private void initialize() {

        // Configura o que cada coluna da tabela deve mostrar
        configurarTabela();

        // Faz com que os campos de ID e ano aceitem apenas números
        configurarCamposNumericos();

        // Configura o comportamento quando selecionamos um livro na tabela
        configurarSelecaoDaTabela();

        // Começa sem nenhuma mensagem
        lblMensagem.setText("");

        // Carrega os livros do banco para a tabela
        carregarLivros();
    }


    // Configura as colunas da tabela
    private void configurarTabela() {

        // A coluna ID vai mostrar o atributo "id" do LivroDTO
        colId.setCellValueFactory(
                new PropertyValueFactory<>("id")
        );

        // A coluna Título vai mostrar o atributo "titulo"
        colTitulo.setCellValueFactory(
                new PropertyValueFactory<>("titulo")
        );

        // A coluna Autor vai mostrar o atributo "autor"
        colAutor.setCellValueFactory(
                new PropertyValueFactory<>("autor")
        );

        // A coluna Ano vai mostrar o atributo "ano_publicacao"
        colAno.setCellValueFactory(
                new PropertyValueFactory<>("ano_publicacao")
        );
    }


    // Configura os campos que devem aceitar apenas números
    private void configurarCamposNumericos() {

        // Observa qualquer alteração feita no campo txtAno
        txtAno.textProperty().addListener(
                (observable, valorAntigo, valorNovo) -> {

                    // Se o novo valor tiver algo que não seja número
                    // OU tiver mais de 4 caracteres
                    if (!valorNovo.matches("\\d*")
                            || valorNovo.length() > 4) {

                        // Volta o campo para o valor anterior
                        txtAno.setText(valorAntigo);
                    }
                }
        );


        // Faz a mesma coisa para o campo ID
        txtId.textProperty().addListener(
                (observable, valorAntigo, valorNovo) -> {

                    // Se o usuário digitar algo que não seja número
                    if (!valorNovo.matches("\\d*")) {

                        // Volta para o valor anterior
                        txtId.setText(valorAntigo);
                    }
                }
        );
    }


    // Configura o que acontece quando o usuário seleciona
    // um livro na tabela
    private void configurarSelecaoDaTabela() {

        // Pega o item selecionado atualmente na tabela
        tabelaLivros.getSelectionModel()
                .selectedItemProperty()

                // Adiciona um "ouvinte" para perceber quando a seleção muda
                .addListener((observable, livroAntigo, livroSelecionado) -> {

                    // Verifica se realmente existe um livro selecionado
                    if (livroSelecionado != null) {

                        // Coloca os dados do livro nos campos da tela
                        preencherCampos(livroSelecionado);
                    }
                });
    }


    // Carrega os livros do banco de dados para a tabela
    @FXML
    private void carregarLivros() {

        // try tenta executar um código que pode gerar erro
        try {

            // Busca todos os livros através do DAO
            // e transforma o resultado em uma lista observável do JavaFX
            ObservableList<LivroDTO> lista =
                    FXCollections.observableArrayList(
                            livroDAO.listarLivros()
                    );

            // Coloca a lista de livros dentro da tabela
            tabelaLivros.setItems(lista);

        } catch (Exception e) {

            // Se acontecer algum erro, mostra uma mensagem
            mostrarMensagemNaTela(
                    "Erro ao carregar os livros.",
                    true
            );

            // Mostra os detalhes do erro no console
            e.printStackTrace();
        }
    }


    // Método executado quando o botão CADASTRAR é clicado
    @FXML
    private void btnCadastrarAction(ActionEvent event) {

        // Pega o texto digitado no campo título
        // trim() remove espaços no começo e no final
        String titulo = txtTitulo.getText().trim();

        // Pega o autor digitado
        String autor = txtAutor.getText().trim();

        // Pega o ano digitado
        String anoTexto = txtAno.getText().trim();


        // Envia os dados para o Validator verificar
        String erro = LivroValidator.validar(
                titulo,
                autor,
                anoTexto
        );


        // Se o Validator encontrou algum erro
        if (erro != null) {

            // Mostra a mensagem de erro
            mostrarMensagemNaTela(erro, true);

            // Para a execução do método
            return;
        }


        // Cria um novo objeto LivroDTO
        LivroDTO livro = new LivroDTO();

        // Coloca o título digitado dentro do objeto
        livro.setTitulo(titulo);

        // Coloca o autor dentro do objeto
        livro.setAutor(autor);

        // Converte o ano de String para int
        // e coloca dentro do objeto
        livro.setAno_publicacao(
                Integer.parseInt(anoTexto)
        );


        try {

            // Envia o livro para o DAO cadastrar no banco
            livroDAO.cadastrarLivro(livro);


            // Mostra mensagem de sucesso
            mostrarMensagemNaTela(
                    "Livro cadastrado com sucesso!",
                    false
            );


            // Atualiza a tabela
            carregarLivros();

            // Limpa os campos
            limparCampos();

        } catch (Exception e) {

            // Se ocorrer algum erro no cadastro
            mostrarMensagemNaTela(
                    "Erro ao cadastrar o livro.",
                    true
            );

            // Mostra o erro no console
            e.printStackTrace();
        }
    }


    // Método executado quando o botão LISTAR é clicado
    @FXML
    private void btnListarAction(ActionEvent event) {

        // Recarrega os livros do banco
        carregarLivros();

        // Mostra mensagem de sucesso
        mostrarMensagemNaTela(
                "Lista de livros atualizada!",
                false
        );
    }


    // Método executado quando o botão BUSCAR é clicado
    @FXML
    private void btnBuscarAction(ActionEvent event) {

        // Verifica se o campo ID está vazio
        if (txtId.getText().isBlank()) {

            // Mostra uma mensagem de erro
            mostrarMensagemNaTela(
                    "Digite um ID para buscar.",
                    true
            );

            // Para o método
            return;
        }


        // Converte o ID de String para int
        int id = Integer.parseInt(txtId.getText());


        try {

            // Procura no banco um livro com aquele ID
            LivroDTO livro = livroDAO.buscarPorId(id);


            // Se não encontrou nenhum livro
            if (livro == null) {

                // Mostra mensagem
                mostrarMensagemNaTela(
                        "Livro não encontrado.",
                        true
                );

                // Para o método
                return;
            }


            // Se encontrou, coloca os dados nos campos
            preencherCampos(livro);


            // Mostra mensagem de sucesso
            mostrarMensagemNaTela(
                    "Livro encontrado!",
                    false
            );

        } catch (Exception e) {

            // Caso aconteça algum erro
            mostrarMensagemNaTela(
                    "Erro ao buscar o livro.",
                    true
            );

            // Mostra o erro no console
            e.printStackTrace();
        }
    }


    // Método executado quando o botão ATUALIZAR é clicado
    @FXML
    private void btnAtualizarAction(ActionEvent event) {

        // Verifica se existe um ID
        if (txtId.getText().isBlank()) {

            // Sem ID não sabemos qual livro atualizar
            mostrarMensagemNaTela(
                    "Selecione ou informe o ID do livro.",
                    true
            );

            return;
        }


        // Pega os dados digitados
        String titulo = txtTitulo.getText().trim();
        String autor = txtAutor.getText().trim();
        String anoTexto = txtAno.getText().trim();


        // Valida os dados
        String erro = LivroValidator.validar(
                titulo,
                autor,
                anoTexto
        );


        // Se houver erro, mostra a mensagem e para
        if (erro != null) {
            mostrarMensagemNaTela(erro, true);
            return;
        }


        // Cria um objeto para representar o livro
        LivroDTO livro = new LivroDTO();

        // Define qual livro será atualizado
        livro.setId(Integer.parseInt(txtId.getText()));

        // Define o novo título
        livro.setTitulo(titulo);

        // Define o novo autor
        livro.setAutor(autor);

        // Define o novo ano
        livro.setAno_publicacao(
                Integer.parseInt(anoTexto)
        );


        try {

            // Envia o livro para o DAO atualizar no banco
            livroDAO.atualizarLivro(livro);


            // Mostra mensagem de sucesso
            mostrarMensagemNaTela(
                    "Dados atualizados com sucesso!",
                    false
            );


            // Atualiza a tabela
            carregarLivros();

            // Limpa os campos
            limparCampos();

        } catch (Exception e) {

            // Mostra mensagem de erro
            mostrarMensagemNaTela(
                    "Erro ao atualizar o livro.",
                    true
            );

            e.printStackTrace();
        }
    }


    // Método executado quando o botão APAGAR é clicado
    @FXML
    private void btnApagarAction(ActionEvent event) {

        // Verifica se existe um ID
        if (txtId.getText().isBlank()) {

            // Se não tiver ID, não sabe qual livro apagar
            mostrarMensagemNaTela(
                    "Digite ou selecione o ID para apagar.",
                    true
            );

            return;
        }


        // Converte o ID de String para int
        int id = Integer.parseInt(txtId.getText());


        try {

            // Pede para o DAO excluir o livro do banco
            livroDAO.deletarLivro(id);


            // Mostra mensagem de sucesso
            mostrarMensagemNaTela(
                    "Livro excluído com sucesso!",
                    false
            );


            // Atualiza a tabela
            carregarLivros();

            // Limpa os campos
            limparCampos();

        } catch (Exception e) {

            // Mostra mensagem de erro
            mostrarMensagemNaTela(
                    "Erro ao excluir o livro.",
                    true
            );

            e.printStackTrace();
        }
    }


    // Método executado quando o botão LIMPAR é clicado
    @FXML
    private void btnLimparAction(ActionEvent event) {

        // Limpa todos os campos
        limparCampos();

        // Mostra mensagem
        mostrarMensagemNaTela(
                "Campos limpos!",
                false
        );
    }


    // Método chamado quando um livro é selecionado
    @FXML
    private void selecionarLivro() {

        // Pega o livro atualmente selecionado na tabela
        LivroDTO livro = tabelaLivros
                .getSelectionModel()
                .getSelectedItem();


        // Verifica se algum livro foi selecionado
        if (livro != null) {

            // Coloca os dados dele nos campos
            preencherCampos(livro);
        }
    }


    // Preenche os campos da tela com os dados de um livro
    private void preencherCampos(LivroDTO livro) {

        // Coloca o ID no campo ID
        txtId.setText(String.valueOf(livro.getId()));

        // Coloca o título no campo título
        txtTitulo.setText(livro.getTitulo());

        // Coloca o autor no campo autor
        txtAutor.setText(livro.getAutor());

        // Coloca o ano no campo ano
        txtAno.setText(
                String.valueOf(livro.getAno_publicacao())
        );

        // Apaga qualquer mensagem anterior
        lblMensagem.setText("");
    }


    // Limpa todos os campos da tela
    private void limparCampos() {

        // Limpa o campo ID
        txtId.clear();

        // Limpa o campo título
        txtTitulo.clear();

        // Limpa o campo autor
        txtAutor.clear();

        // Limpa o campo ano
        txtAno.clear();

        // Remove a seleção da tabela
        tabelaLivros.getSelectionModel().clearSelection();
    }


    // Mostra mensagens para o usuário
    // ehErro = true  -> mensagem de erro
    // ehErro = false -> mensagem de sucesso
    private void mostrarMensagemNaTela(
            String texto,
            boolean ehErro
    ) {

        // Coloca o texto recebido no Label
        lblMensagem.setText(texto);


        // Verifica se é uma mensagem de erro
        if (ehErro) {

            // Deixa o texto vermelho e em negrito
            lblMensagem.setStyle(
                    "-fx-text-fill: #cc0000; "
                            + "-fx-font-weight: bold;"
            );

        } else {

            // Deixa o texto verde e em negrito
            lblMensagem.setStyle(
                    "-fx-text-fill: #008000; "
                            + "-fx-font-weight: bold;"
            );
        }
    }
}
